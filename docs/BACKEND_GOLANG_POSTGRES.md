# CFit Talent — Golang + PostgreSQL Backend Architecture

**Stack:** Go 1.22+ · PostgreSQL 16 · Chi / Gin · sqlc or pgx · golang-migrate · JWT  
**Companion docs:** [`BACKEND_API_GUIDELINE.md`](./BACKEND_API_GUIDELINE.md) (HTTP contracts) · this file (Go project layout)  
**Roles:** `admin` | `player` | `coach` | `scout` — **scouts are admin-invited only**

---

## 1. Recommended stack

| Layer | Choice | Why |
|-------|--------|-----|
| HTTP | `chi` or `gin` | Middleware-friendly, production common |
| DB driver | `pgx/v5` + `sqlc` | Type-safe queries, fast |
| Migrations | `golang-migrate` | Versioned SQL |
| Auth | JWT (access) + hashed refresh optional | Matches frontend Bearer |
| Password | `bcrypt` or `argon2id` | |
| Config | `envconfig` / `viper` | |
| Validation | `go-playground/validator` | |
| Logging | `slog` or `zap` | |
| Docs | `swaggo/swag` → OpenAPI | |
| Email | SMTP / Resend / SES adapter | Invites, reset, org approval |
| Files | Local disk → S3-compatible later | Player photos |
| Jobs (optional) | `asynq` + Redis | Emails, reminders |

**Layer rules**

- `handler` → `service` → `repository` → PostgreSQL  
- `domain` has no import of `chi` / `pgx`  
- Scout register endpoint **does not exist** in `auth_handler` — only admin invite in `admin_handler`

## 4. PostgreSQL schema (core)

Use `UUID` PKs, `timestamptz`, soft-delete only where needed. Status columns stay **independent**.

```sql
-- roles
CREATE TYPE user_role AS ENUM ('admin', 'player', 'coach', 'scout');

CREATE TABLE users (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email         CITEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  full_name     TEXT NOT NULL,
  phone         TEXT,
  role          user_role NOT NULL,
  is_active     BOOLEAN NOT NULL DEFAULT true,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TYPE org_status AS ENUM ('pending', 'under_review', 'approved', 'rejected');
CREATE TYPE org_type AS ENUM (
  'academy', 'club', 'school', 'league', 'federation', 'independent_coach'
);

CREATE TABLE organizations (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name            TEXT NOT NULL,
  type            org_type NOT NULL,
  country         TEXT NOT NULL,
  city            TEXT NOT NULL,
  status          org_status NOT NULL DEFAULT 'pending',
  website         TEXT,
  social          TEXT,
  age_groups      TEXT,
  player_count_band TEXT,
  director_name   TEXT NOT NULL,
  director_role   TEXT NOT NULL,
  email           CITEXT NOT NULL,
  phone           TEXT NOT NULL,
  additional_contact TEXT,
  invite_slug     TEXT UNIQUE,          -- for coach invite links
  approved_at     TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE organization_users (
  organization_id UUID REFERENCES organizations(id),
  user_id         UUID REFERENCES users(id),
  PRIMARY KEY (organization_id, user_id)
);

CREATE TYPE profile_status AS ENUM (
  'registered', 'incomplete', 'complete', 'verified', 'review_ready'
);
CREATE TYPE intelligence_stage AS ENUM (
  'submitted', 'verification', 'watch', 'review', 'priority', 'pass', 'archived'
);
CREATE TYPE verification_status AS ENUM ('pending', 'ready_to_review', 'verified');

CREATE TABLE players (
  id                   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id              UUID UNIQUE NOT NULL REFERENCES users(id),
  cfit_id              TEXT UNIQUE,              -- e.g. CFT-2026-JM-0042
  organization_id      UUID REFERENCES organizations(id), -- via invite
  date_of_birth        DATE NOT NULL,
  country              TEXT NOT NULL,
  nationality          TEXT NOT NULL,
  city                 TEXT,
  height_cm            NUMERIC(5,1),
  weight_kg            NUMERIC(5,1),
  preferred_foot       TEXT,
  primary_position     TEXT,
  secondary_position   TEXT,
  photo_url            TEXT,
  profile_status       profile_status NOT NULL DEFAULT 'registered',
  verification         verification_status NOT NULL DEFAULT 'pending',
  intelligence         intelligence_stage NOT NULL DEFAULT 'submitted',
  scout_visibility     BOOLEAN NOT NULL DEFAULT false,
  review_ready         BOOLEAN NOT NULL DEFAULT false,
  -- football / video / contact as JSONB or separate tables
  football             JSONB NOT NULL DEFAULT '{}',
  videos               JSONB NOT NULL DEFAULT '{}',
  contacts             JSONB NOT NULL DEFAULT '{}',  -- never expose to scout APIs
  created_at           TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at           TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE scouts (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID UNIQUE REFERENCES users(id), -- null until invite accepted
  scout_code      TEXT UNIQUE,                       -- SC-101
  email           CITEXT NOT NULL UNIQUE,
  full_name       TEXT NOT NULL,
  organization    TEXT NOT NULL,
  title           TEXT,
  invitation      TEXT NOT NULL DEFAULT 'pending',   -- pending | accepted
  account_status  TEXT NOT NULL DEFAULT 'invitation_pending',
  invite_token    TEXT UNIQUE,
  invite_expires  TIMESTAMPTZ,
  last_active_at  TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Append-only sharing audit
CREATE TABLE player_share_events (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  player_id   UUID NOT NULL REFERENCES players(id),
  scout_id    UUID NOT NULL REFERENCES scouts(id),
  event_type  TEXT NOT NULL, -- shared | revoked
  actor_id    UUID NOT NULL REFERENCES users(id),
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE player_shares (
  player_id   UUID REFERENCES players(id),
  scout_id    UUID REFERENCES scouts(id),
  shared_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (player_id, scout_id)
);

CREATE TABLE scout_watchlist (
  scout_user_id UUID REFERENCES users(id),
  player_id     UUID REFERENCES players(id),
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (scout_user_id, player_id)
);

CREATE TABLE scout_reviews (
  id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  scout_user_id UUID NOT NULL REFERENCES users(id),
  player_id    UUID NOT NULL REFERENCES players(id),
  action       TEXT NOT NULL,
  topic        TEXT,
  message      TEXT,
  status       TEXT NOT NULL, -- under_review | responded | completed | recorded
  admin_response TEXT,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE notifications (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id    UUID NOT NULL REFERENCES users(id),
  type       TEXT NOT NULL,
  title      TEXT NOT NULL,
  body       TEXT NOT NULL,
  unread     BOOLEAN NOT NULL DEFAULT true,
  meta       JSONB NOT NULL DEFAULT '{}',
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE milestone_records (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  player_id     UUID NOT NULL REFERENCES players(id),
  milestone_key TEXT NOT NULL, -- milestone1 | milestone2
  presented_at  DATE,
  review_ready_at DATE,
  delivery_status TEXT DEFAULT 'pending', -- pending | reviewed | forwarded
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE activity_logs (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  entity_type TEXT NOT NULL,
  entity_id  UUID NOT NULL,
  tag        TEXT,
  title      TEXT NOT NULL,
  subtitle   TEXT,
  actor_id   UUID REFERENCES users(id),
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
```

Add indexes on `players.organization_id`, `player_shares.scout_id`, `scout_reviews.status`, `notifications(user_id, unread)`.

## 5. Router sketch (`internal/server/router.go`)

```go
r := chi.NewRouter()
r.Use(middleware.RequestID, middleware.RealIP, logging, recoverer, cors)

r.Route("/api", func(api chi.Router) {
  // Public
  api.Post("/auth/login", h.Auth.Login)
  api.Post("/auth/register/player", h.Auth.RegisterPlayer)
  api.Post("/auth/register/organization", h.Auth.RegisterOrganization)
  api.Get("/auth/scout-invite/{token}", h.Auth.ScoutInvitePreview)
  api.Post("/auth/scout-invite/{token}/accept", h.Auth.ScoutInviteAccept)
  api.Post("/auth/forgot-password", h.Auth.ForgotPassword)
  api.Post("/auth/reset-password", h.Auth.ResetPassword)
  api.Get("/public/invite/{slug}", h.Public.InviteSlug)
  api.Post("/public/partnership-inquiries", h.Public.Partnership)

  api.Group(func(pr chi.Router) {
    pr.Use(authMiddleware) // JWT required

    pr.Get("/auth/me", h.Auth.Me)
    pr.Post("/auth/logout", h.Auth.Logout)
    pr.Get("/users/me", h.Users.GetMe)
    pr.Patch("/users/me", h.Users.UpdateMe)
    pr.Post("/users/me/password", h.Users.ChangePassword)
    pr.Get("/notifications", h.Notify.List)
    // ... mark read

    pr.Route("/players/me", func(pl chi.Router) {
      pl.Use(requireRole("player"))
      // dashboard, sections, photo
    })

    pr.Route("/coach", func(c chi.Router) {
      c.Use(requireRole("coach"), requireApprovedOrg)
      // dashboard, org, players, invite, recommendations, milestones
    })

    pr.Route("/scout", func(s chi.Router) {
      s.Use(requireRole("scout"), requireActiveScout)
      // dashboard, players, watchlist, actions, reviews, assigned
    })

    pr.Route("/admin", func(a chi.Router) {
      a.Use(requireRole("admin"))
      // overview, players, orgs, scouts invitations, share, feedback, milestones
      a.Post("/scouts/invitations", h.Admin.InviteScout) // ONLY way to create scout
    })
  })
})
```

**Forbidden:** any route like `POST /auth/register/scout`.
